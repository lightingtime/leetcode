public ListNode addTwoNumbers(ListNode l1, ListNode l2) {
        // TODO: 在这里实现你的解法
        ListNode dummy = new ListNode();
        ListNode p1 = l1, p2 = l2, p = dummy;
        int next = 0;
        while (p1 != null && p2 != null) {
            int sum = p1.val + p2.val + next;
            ListNode node = new ListNode(sum % 10);
            next = sum / 10;
            p.next = node;
            p = p.next;
            p1 = p1.next;
            p2 = p2.next;
        }
        while (p1 != null) {
            int sum = p1.val + next;
            ListNode node = new ListNode(sum % 10);
            next = sum / 10;
            p.next = node;
            p = p.next;
            p1 = p1.next;
        }
        while (p2 != null) {
            int sum = p2.val + next;
            ListNode node = new ListNode(sum % 10);
            next = sum / 10;
            p.next = node;
            p = p.next;
            p2 = p2.next;
        }
        if (next != 0) {
            p.next = new ListNode(next);
        }
        return dummy.next;
    }
