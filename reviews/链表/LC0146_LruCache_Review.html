static class LRUCache {
        Map<Integer, DoubleLinkNode> map;
        DoubleLinkNode head;
        DoubleLinkNode tail;
        int size;
        int capacity;
        public LRUCache(int capacity) {
            this.capacity = capacity;
            size = 0;
            map = new HashMap<>();
            head = new DoubleLinkNode();
            tail = new DoubleLinkNode();
            head.next = tail;
            tail.pre = head;
        }
        public int get(int key) {
            if (!map.containsKey(key)) {
                return -1;
            }
            DoubleLinkNode cur = map.get(key);
            cutNode(cur);
            moveToHead(cur);
            return cur.value;
        }

        private void cutNode(DoubleLinkNode cur) {
            if (cur.value == null) {
                return;
            }
            cur.pre.next = cur.next;
            cur.next.pre = cur.pre;
        }

        private void moveToHead(DoubleLinkNode cur) {
            cur.pre = head;
            cur.next = head.next;
            head.next.pre = cur;
            head.next = cur;
        }

        public void put(int key, int value) {
            if (map.containsKey(key)) {
                DoubleLinkNode cur = map.get(key);
                cur.key = key;
                cur.value = value;
                cutNode(cur);
                moveToHead(cur);
            } else {
                DoubleLinkNode cur = new DoubleLinkNode();
                moveToHead(cur);
                cur.key = key;
                cur.value = value;
                map.put(key, cur);
                size++;
                if (this.size > this.capacity) {
                    map.remove(tail.pre.key);
                    cutNode(tail.pre);
                    size--;
                }
            }
        }

        class DoubleLinkNode {
            DoubleLinkNode next;
            DoubleLinkNode pre;
            Integer key;
            Integer value;

            public DoubleLinkNode() {
            }

            public DoubleLinkNode(DoubleLinkNode next, DoubleLinkNode pre, int value) {
                this.next = next;
                this.pre = pre;
                this.value = value;
            }
        }
    }

哈希表+双向链表标准设计：插入要 map.put+挂链+size++，淘汰要 map.remove+断链+size--，三者同步；本地三轮修复（map.put 遗漏、普通插入分支为空、size 未维护）后全绿，力扣 25/25 Accepted（O(1) 时间/O(capacity) 空间，已最优）

套路沉淀

LRU 哈希表+双向链表：两结构必须同步增删，哨兵判身份不判值

map 负责 O(1) 定位节点、链表维护使用顺序，两者必须同步：插入新 key 要 map.put(key,node) 并 size++；淘汰时 cut 掉尾节点后，必须用该节点存的 key 做 map.remove 并 size--；get/更新命中时先 cut 再 moveToHead 刷新顺序。哨兵节点用 cur==head || cur==tail 判身份，不要用 value==null 猜。出处：LC0146 lru-cache。
